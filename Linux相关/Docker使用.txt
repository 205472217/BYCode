===============================Docker安装=================================
1.更新软件包索引，并且安装必要的依赖软件
sudo apt update
sudo apt install apt-transport-https ca-certificates curl gnupg-agent software-properties-common

2.添加镜像源公钥、添加软件源（公钥和软件源地址要匹配，此处使用阿里云）
curl -fsSL https://mirrors.aliyun.com/docker-ce/linux/ubuntu/gpg | sudo apt-key add -
sudo add-apt-repository "deb [arch=amd64] https://mirrors.aliyun.com/docker-ce/linux/ubuntu $(lsb_release -cs) stable"

3.更新软件源，并查看软件源中存在的Docker版本
sudo apt update
apt list -a docker-ce

4.选择指定版本的Docker进行安装
# 替换为你实际需要的版本号
sudo apt install docker-ce=<VERSION> docker-ce-cli=<VERSION> containerd.io
比如：
sudo apt install docker-ce=5:20.10.24~3-0~ubuntu-focal docker-ce-cli=5:20.10.24~3-0~ubuntu-focal containerd.io

5.禁止Docker版本更新
sudo apt-mark hold docker-ce

6.允许非root用户执行docker相关指令
sudo usermod -aG docker $USER
newgrp docker

7.检查Docker是否正常安装
docker -v  

==============================Docker使用=======================================
Docker需要拉取镜像文件，在本地运行起来（一个实例即一个容器）

1.配置镜像源（不能有#标注的注释，必须原样写入daemon.json)
sudo tee /etc/docker/daemon.json <<-'EOF'
{
  "registry-mirrors": [
    "https://docker.xuanyuan.me",
    "https://docker.mirrors.ustc.edu.cn",
    "https://hub-mirror.c.163.com",
    "https://mirror.aliyuncs.com"
  ]
}
EOF

2.验证镜像拉取环境是否正常
docker run hello-world			#hello-world镜像文件只有几KB，用于测试

注意：如果拉取环境异常，可先尝试以下指令后重试，不行的话使用手动下载镜像的方式

# 先停止 docker.service
sudo systemctl stop docker
# 再停止 docker.socket（关键：消除警告，避免自动重启）
sudo systemctl stop docker.socket
# 查看 docker.service 状态（inactive (dead)即可）
sudo systemctl status docker
# 查看 docker.socket 状态（inactive (dead)即可）
sudo systemctl status docker.socket


# 先启动 docker.socket
sudo systemctl start docker.socket
# 再启动 docker.service
sudo systemctl start docker
# 验证启动状态（Active 为 running 即正常）
sudo systemctl status docker

3.下载ubuntu镜像，20.04为长期支持版本
docker pull ubuntu:20.04


4.手动下载镜像，并导入
# 进入tar包所在位置，导入镜像（核心命令）
docker load < ubuntu-20.04.tar

# 查看可用镜像
docker images
# 删除指定镜像
docker rmi <镜像名>

#保存容器为镜像
docker commit runreleaseenv image:latest

#打包镜像为tar包
docker save -o /root/img_docker.tar img:latest

#容器改名
docker rename <old镜像名> <new镜像名>

#容器间的目录拷贝
docker exec <源容器> tar -cf - -C <源父目录> <待拷贝目录名> | docker exec -i <目标容器> tar -xf - -C <目标父目录>
======分析服务拷贝
docker exec runreleaseenv tar -cf - -C /opt ASApplication | docker exec -i runreleaseenv_test tar -xf - -C /opt
======抽帧服务拷贝
docker exec runreleaseenv tar -cf - -C /opt extrackFrame | docker exec -i runreleaseenv_test_child tar -xf - -C /opt


5.运行

（创建新的容器实例）
# 格式：docker run -d --name 容器名 镜像名:版本 持续运行的命令

==========支持GPU
docker run -d --name runreleaseenv --cgroupns=host --gpus all -it img:latest bash

==========支持GPU 、 限制内存使用
docker run -d --name runreleaseenv --cgroupns=host --gpus all --memory 15g --memory-swap 15g -it img:latest bash


验证CUDA可用,容器内执行： 
nvidia-smi


注意：如果要映射端口，让外部直接访问Docker，可添加 -p 宿主机端口:容器端口
比如容器内安装SSH，外部连接10222端口即可连接到容器的22端口，则使用 -p 10022:22


		-d：后台运行容器（detach 模式），容器启动后不会占用终端
		--name ubuntu-20.04-bg：容器的自定义名字
		ubuntu:20.04：指定要运行的镜像Tag（版本必须和你导入的一致，docker images ubuntu可查看镜像的Tag）
		tail -f /dev/null：让容器后台持续运行（Ubuntu 镜像默认启动 bash 会立刻退出，这个命令能让容器一直挂起）
（启动之前停止运行的实例）
# 格式：docker start 容器名
docker start ubuntu-20.04-bg 
	容器名可通过 docker ps -a 查看

6.停止
# 格式：docker stop 容器名/容器ID
docker stop ubuntu-20.04-bg

# 停止+删除（二合一）
docker rm -f ubuntu-20.04-bg

7.进入容器环境
docker exec -it ubuntu-20.04-bg /bin/bash

==拷贝Docker的文件==
a.从Docker拷贝到宿主机
	docker cp 容器ID:/app/logs/error.log /tmp/error.log
	
b.从宿主机拷贝到Windows
scp 用户名@LinuxIP:Linux文件路径 Windows本地路径
	scp root@192.168.0.177:/tmp/error.log D:\docker_files\error.log		【在目标主机执行】
	scp /root/error.log	root@192.168.3.156:/root 						【在源主机执行】
	scp /root/error.log	root@192.168.3.156:/root 						【在源主机执行】


==开机自启==
# 设置Docker服务开机自启
sudo systemctl enable docker
# 设置容器开机自启
docker run -d --name 容器名/容器ID --restart=unless-stopped nginx
# 修改已有容器重启策略（替换成你的容器名/ID）
docker update --restart=unless-stopped 容器名/容器ID
	--restart参数：
	unless-stopped		除非手动执行docker stop停止容器，否则重启
	always				始终重启（只要容器退出了）
	on-failure			容器异常退出时重启
	
==迁移镜像==
# 1. 基于原容器做镜像
docker commit ubuntu-20.04-bg ubuntu-20.04:v1	
# 2. 打包镜像为tar包
docker save -o /root/public/ubuntu-20.04.tar ubuntu-20.04:v1
# 3. 另一台机器B加载tar包
docker load < ubuntu-20.04.tar
# 4. 此时机器B使用docker images即可查看到镜像ubuntu-20.04已存在，TAG为v1 
	
==容器互联==
# 格式：docker network create --driver bridge 自定义网络名
docker network create --driver bridge my-network
# 验证：查看创建的网络
docker network ls
# a.已运行的容器加入网络
# 格式：docker network connect 网络名 容器名
docker network connect my-network 已运行的容器名
# b.启动容器时加入网络
Docker run 时指定 --network 网络名
