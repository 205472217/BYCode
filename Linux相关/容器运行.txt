【拷贝程序文件夹】
=======执行前保证容器已经创建，容器名正确=======
#拷贝 <192.168.1.127> 上 runreleaseenv:/opt/extrackFrame ==> <192.168.3.156> 上 runreleaseenv_2:/opt/extrackFrame

1.在<源宿主机> 执行（先压缩，再传输）
docker exec -i runreleaseenv sh -c "cd /opt && tar -cf - extrackFrame" > extrackFrame.tar
scp extrackFrame.tar root@192.168.3.156:/root/

2.在<目标宿主机执行>
cat /root/extrackFrame.tar | docker exec -i runreleaseenv_2 tar -xpf - -C /opt

================================运行环境==================================
【extrackFrame】
[===解压aws库===]
tar -zxvf aws_sdk_all.tar.gz -C /usr/local

[===下载aria2===]
apt update && apt install -y aria2

[===创建log_data文件夹===]
mkdir -p /log_data
chmod 777 /log_data

【ASApplication/ChildApp】
[===创建ramdisk文件夹===]
mkdir -p /ramdisk
chmod 777 /ramdisk

[===指定显卡序号，避免始终使用显卡0导致GPU不足（适用于128多显卡环境）===]
echo 'export CUDA_VISIBLE_DEVICES=1' >> /etc/profile
echo 'export CUDA_MODULE_LOADING=LAZY' >> /etc/profile
source /etc/profile

echo 'export CUDA_VISIBLE_DEVICES=1' >> ~/.bashrc
echo 'export CUDA_MODULE_LOADING=LAZY' >> ~/.bashrc
source ~/.bashrc

==============================查看日志==============================
tail -n 10 -f filename.txt 					（先显示最后 10 行，再开始实时追踪）
sed -n "10,20p" filename.txt				（显示 10行-20行 的内容）


==============================搜索内容==============================
grep -ni "指定内容" log.20.txt log.21.txt		（在指定文件搜索）
grep -rni "指定内容" . --include="*.txt"		（递归文件夹搜索，需指定文件后缀）
grep -C 10 "指定内容" log.1.txt				 	（显示前后10行）

grep -ni "2051903457730854914" aslog.1.txt

==============================清理日志文件(只保留log.1.txt)===================
find . -maxdepth 1 -type f -name "log.*.txt" ! -name "log.1.txt" -delete

set print thread-events off

【3.程序崩溃复现后执行】
bt full
thread apply all bt full
info locals
quit

==============================177机器VS Code无法连接====================================
1.确保容器已启动

2.CMD 执行下列命令，没有提示输密码，一般是容器的ssh服务没启动
ssh root@192.168.0.177 -p 10022 

2.容器内执行下列命令，启动ssh服务即可
mkdir -p /run/sshd
chmod 700 /run/sshd
chown root:root /run/sshd
/usr/sbin/sshd
